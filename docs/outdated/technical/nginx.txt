# This server block handles HTTPS traffic and proxies to your application
server {
    server_name bserver.resomix.com;

    # Allow large uploads (you had 105M)
    client_max_body_size 105M;

    # -----------------------
    # Security response headers
    # -----------------------
    # OWASP secure headers guidance
    # https://cheatsheetseries.owasp.org/cheatsheets/HTTP_Headers_Cheat_Sheet.html
    add_header X-Content-Type-Options nosniff always;
    add_header X-Frame-Options DENY always;
    add_header Referrer-Policy no-referrer-when-downgrade always;
    # If you already force HTTPS everywhere, enable HSTS (subdomains optional)
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

    # -----------------------
    # Global safety net limits
    # -----------------------
    # Apply a gentle global cap to everything on this host
    limit_req zone=global_rl burst=10;      # allows short spikes, then enforces 5 rps avg
    limit_conn perip_conn 30;               # total concurrent conns per IP

    # ------- Minimal CORS for NGINX-generated responses --------
    # Let FastAPI handle CORS for successful proxied responses.
    set $cors_methods "GET, POST, PUT, PATCH, DELETE, OPTIONS";
    set $cors_headers "Authorization, Content-Type, X-Requested-With, X-Api-Key";
    set $cors_expose  "Content-Length, Content-Range, Accept-Ranges, Content-Disposition";

    # Only reflect Origin if it's on the allowlist from the http-level map


    # -----------------------
    # API + app proxy
    # -----------------------
    # Tighter limits for API-like paths; you can refine prefixes as needed.
    location ~ ^/(api|search|upload|v1|v2)/ {
        # API/search/db operations: your target is ~3 rps + short burst
        # https://nginx.org/en/docs/http/ngx_http_limit_req_module.html
        limit_req zone=api_rl burst=10 nodelay;
        # Keep per-IP concurrent sockets reasonable for app endpoints
        # https://nginx.org/en/docs/http/ngx_http_limit_conn_module.html
        limit_conn perip_conn 20;

        proxy_pass http://127.0.0.1:8001;
        proxy_http_version 1.1;

        # WebSocket & proxy headers
        # https://nginx.org/en/docs/http/websocket.html
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;

        # Preserve real client details (Cloudflare real IP already restored at http-level)
        proxy_set_header X-Real-IP       $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout  60s;
        proxy_send_timeout    300s;
        proxy_read_timeout    300s;

        proxy_cache_bypass $http_upgrade;
    }

    # Default catch-all (non-API paths). Keeps your previous behavior but benefits from global limits.
    location / {
        proxy_pass http://127.0.0.1:8001;
        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;

        proxy_set_header X-Real-IP       $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout  60s;
        proxy_send_timeout    300s;
        proxy_read_timeout    300s;

        proxy_cache_bypass $http_upgrade;
    }

    # ---- TLS (Certbot-managed) ----
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/bserver.resomix.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/bserver.resomix.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

    # Optional: per-site logs
    # access_log /var/log/nginx/bserver.resomix.com.access.log;
    # error_log  /var/log/nginx/bserver.resomix.com.error.log;
}

# This server block handles HTTP traffic and redirects it to HTTPS
server {
    client_max_body_size 105M;

    if ($host = bserver.resomix.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    server_name bserver.resomix.com;
    return 404; # managed by Certbot
}
