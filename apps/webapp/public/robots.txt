# -----------------------------------------------------
# robots.txt for Resomix Beta (https://beta.resomix.com)
#
# Crawling is ALLOWED across the app, and almost every page carries a `noindex` robots meta tag
# instead of being blocked here. This app is a service rather than a set of pages to be found: the
# words belong on resomix.com, and `/auth/login` is the single address that stays indexable,
# because it is the way in and the result someone searching for the sign-in page wants.
#
# That distinction matters: a page blocked in robots.txt is never fetched, so its noindex tag is
# never read, and the URL can still surface in results as a bare link. Allow the crawl and let the
# tag do the work.
#
# This file previously disallowed /*/ and /*? — blocking every page except the root — while
# index.html separately declared noindex,nofollow and a sitemap advertised four URLs. Three
# contradictory signals; they now agree.
# -----------------------------------------------------

User-agent: *
Allow: /

# Build artefacts and source maps — never useful, occasionally revealing.
Disallow: /*.map$
Disallow: /node_modules/
Disallow: /.git/
Disallow: /.turbo/

# Transient upload screen: no content, and every visit is a different in-flight job.
Disallow: /loader

# Static assets crawlers need to render the app.
Allow: /fonts/
Allow: /*.css$
Allow: /*.js$

Sitemap: https://beta.resomix.com/sitemap.xml
